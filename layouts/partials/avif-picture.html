{{- $res := .res -}} {{- $alt := .alt | default "" -}} {{- $extra := .attrs | default dict -}} {{- $isAvif := eq $res.MediaType.SubType "avif" -}} {{- $isOrig := .orig | default false -}} {{- if or $isAvif $isOrig -}} {{- $jpegW := int (strings.TrimSuffix "x" (.size | default "1600x")) -}} {{- if lt $res.Width $jpegW }}{{ $jpegW = $res.Width }}{{ end -}} {{- $avifW := int (strings.TrimSuffix "x" (.avifSize | default "2400x")) -}} {{- if lt $res.Width $avifW }}{{ $avifW = $res.Width }}{{ end -}} {{- $source := "" -}} {{- if $isOrig -}} {{- $source = ($res.Process (printf "resize %dx avif" $avifW)).RelPermalink -}} {{- else -}} {{- $source = $res.RelPermalink -}} {{- end -}} {{- $fb := $res.Process (printf "resize %dx jpeg" $jpegW) -}} {{- $s := "" -}} {{- range $k, $v := merge $extra (dict "src" $fb.RelPermalink "alt" $alt "loading" "lazy") -}} {{- if $v }}{{ $s = printf "%s %s=%q" $s $k $v }}{{ end -}} {{- end -}} {{- else -}} {{- $s := "" -}} {{- range $k, $v := merge $extra (dict "src" $res.RelPermalink "alt" $alt "loading" "lazy") -}} {{- if $v }}{{ $s = printf "%s %s=%q" $s $k $v }}{{ end -}} {{- end -}} {{- end -}}
